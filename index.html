<script>

  const tg = window.Telegram?.WebApp; try { tg?.expand(); } catch(e){}

  const $ = id => document.getElementById(id);

  const money = x => (Number.isFinite(x)?x:0).toLocaleString('sv-SE',{minimumFractionDigits:2,maximumFractionDigits:2});

  const clamp = (x, min) => Math.max(min, x);


  function compute(o1, o2, S){

    if(o1<=1 || o2<=1 || S<=0) return {s1:0,s2:0,profit:0,roi:0,isArb:false};

    const s1 = S * (o2/(o1+o2));

    const s2 = S * (o1/(o1+o2));

    const payout = s1*o1;

    const profit = payout - S;

    const invsum = 1/o1 + 1/o2;

    return { s1, s2, profit, roi: profit/S, isArb: invsum < 1.0 };

  }


  function render(){

    const o1 = parseFloat($('o1').value.toString().replace(',', '.'))||0;

    const o2 = parseFloat($('o2').value.toString().replace(',', '.'))||0;

    const S  = parseFloat($('stake').value.toString().replace(',', '.'))||0;

    const r  = compute(o1,o2,S);

    const status = r.isArb ? '<span class="ok">✅ Arbitrage</span>' : '<span class="warn">⚠️ Not an arb</span>';

    $('out').innerHTML = `

      <div>Odds: <span class="fine">${o1.toFixed(2)}</span> / <span class="fine">${o2.toFixed(2)}</span></div>

      <div class="kpi"><span>Stake ${$('l1').textContent || 'Side 1'}</span><strong>${money(r.s1)} kr</strong></div>

      <div class="kpi"><span>Stake ${$('l2').textContent || 'Side 2'}</span><strong>${money(r.s2)} kr</strong></div>

      <div class="kpi"><span>Profit</span><strong>${money(r.profit)} kr</strong></div>

      <div class="kpi"><span>ROI</span><strong>${(r.roi*100).toFixed(2)}%</strong></div>

      <div class="status">${status}</div>

    `;

  }


  ['o1','o2','stake'].forEach(id=>{

    $(id).addEventListener('input', render);

    $(id).addEventListener('change', e=>{

      const min = id==='stake'?1:1.01;

      const v = parseFloat(e.target.value)||min;

      e.target.value = clamp(v, min).toFixed(2);

      render();

    });

  });


  document.addEventListener('click', e=>{

    const btn = e.target.closest('button[data-step]');

    if(!btn) return;

    const step = parseFloat(btn.dataset.step);

    const target = $(btn.dataset.target);

    let v = parseFloat(target.value.toString().replace(',', '.')) || 0;

    v += step;

    const min = target.id==='stake'?1:1.01;

    target.value = clamp(v, min).toFixed(2);

    render();

  });


  // Prefill from query string: o1, o2, S, l1, l2, evt, mkt

  const p = new URL(location.href).searchParams;

  const qsO1 = p.get('o1'), qsO2 = p.get('o2'), qsS = p.get('S');

  const l1 = p.get('l1'), l2 = p.get('l2');

  const evt = p.get('evt'), mkt = p.get('mkt');

  if(l1) $('l1').textContent = l1;

  if(l2) $('l2').textContent = l2;

  if(evt || mkt) document.getElementById('meta').textContent = `${evt?evt:''}${evt&&mkt?' • ':''}${mkt?mkt:''}`;

  if(qsO1) $('o1').value = qsO1;

  if(qsO2) $('o2').value = qsO2;

  if(qsS)  $('stake').value = qsS;


  render();

</script>
